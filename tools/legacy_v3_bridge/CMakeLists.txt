cmake_minimum_required(VERSION 3.29)
project(YsmLegacyV3 LANGUAGES CXX)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_subdirectory("${YSM_NATIVE_SOURCE}" native EXCLUDE_FROM_ALL)
add_library(YsmLegacyV3 SHARED bridge.cpp)
target_include_directories(YsmLegacyV3 PRIVATE "${YSM_JSON_INCLUDE}"
    "${YSM_NATIVE_SOURCE}/modules/legacy/src")
target_link_libraries(YsmLegacyV3 PRIVATE ysm-legacy)
if(MSVC)
    target_compile_options(ysm-legacy PRIVATE /bigobj)
    target_compile_options(YsmLegacyV3 PRIVATE /utf-8 /bigobj)
endif()
